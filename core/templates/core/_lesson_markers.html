const lessonMarkerColors={yellow:'#fde047',green:'#86efac',blue:'#93c5fd',pink:'#f9a8d4',purple:'#d8b4fe',orange:'#fdba74',red:'#fca5a5'}; function toggleLessonMarkers(existing, selected, color){ const same=(a,b)=>Number(a.block_id)===Number(b.block_id)&&a.text===b.text; const erase=selected.every(s=>{ let covered=s.start; existing.filter(r=>same(r,s)).sort((a,b)=>a.start-b.start).forEach(r=>{ if(r.start<=covered && r.end>covered)covered=r.end; }); return covered>=s.end; }); let result=existing.map(r=>({...r,color:r.color||'yellow'})); selected.forEach(s=>{ result=result.flatMap(r=>{ if(!same(r,s)||r.end<=s.start||r.start>=s.end)return [r]; return [...(r.starts.end?[{...r,start:s.end}]:[])]; }); }); if(!erase)result.push(...selected.map(r=>({...r,color}))); result.sort((a,b)=>a.block_id-b.block_id||a.start-b.start); const merged=[]; result.forEach(r=>{const last=merged.at(-1);if(last&&same(last,r)&&last.color===r.color&&r.start<=last.end)last.end=Math.max(last.end,r.end);else merged.push(r)}); return merged; } function applyLessonMarkers(ranges){ const box = qs("moduleBlocks"); const signature = JSON.stringify(ranges || []); if (box.dataset.markers === signature) return; box.dataset.markers = signature; box.querySelectorAll(".lesson-marker").forEach(mark => mark.replaceWith(document.createTextNode(mark.textContent))); box.normalize(); box.querySelectorAll('[data-block-type="text"],[data-block-type="heading"],[data-block-type="code"]').forEach(block => { const content = block.dataset.blockType === "code" ? block.querySelector("pre") : block; const text = content.textContent; const parts = (ranges || []).filter(r => Number(r.block_id) === Number(block.dataset.blockId) && r.text === text) .sort((a,b) => a.start-b.start); const merged = []; parts.forEach(r => { const last = merged.at(-1); const color=r.color||'yellow'; if (last && color===last.color && r.start <= last.end) last.end = Math.max(last.end, r.end); else merged.push({start:r.start,end:r.end,color}); }); if (!merged.length) return; const fragment = document.createDocumentFragment(); let offset = 0; merged.forEach(r => { fragment.append(document.createTextNode(text.slice(offset,r.start))); const mark = document.createElement("mark"); mark.className = "lesson-marker"; mark.style.setProperty('--marker-color',lessonMarkerColors[r.color]||lessonMarkerColors.yellow); mark.textContent = text.slice(r.start,r.end); fragment.append(mark); offset = r.end; }); fragment.append(document.createTextNode(text.slice(offset))); content.replaceChildren(fragment); }); } function selectedLessonRanges(){ const selection = window.getSelection(); if (!selection || selection.isCollapsed || !selection.rangeCount) return []; const range = selection.getRangeAt(0); const box = qs("moduleBlocks"); if (!box.contains(range.startContainer) || !box.contains(range.endContainer)) return []; const rows = []; box.querySelectorAll('[data-block-type="text"],[data-block-type="heading"],[data-block-type="code"]').forEach(block => { const content = block.dataset.blockType === "code" ? block.querySelector("pre") : block; if (!range.intersectsNode(content)) return; const before = document.createRange(); before.selectNodeContents(content); let start = 0, end = content.textContent.length; if (content.contains(range.startContainer)) { before.setEnd(range.startContainer, range.startOffset); start = before.toString().length; } if (content.contains(range.endContainer)) { before.selectNodeContents(content); before.setEnd(range.endContainer, range.endOffset); end = before.toString().length; } if (end > start) rows.push({block_id:Number(block.dataset.blockId),start,end,text:content.textContent}); }); return rows; }
